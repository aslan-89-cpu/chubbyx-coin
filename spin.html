// Array configurations for your prize segments const spinPrizes = [ { text: "5,000", value: 5000, minAngle: 0, maxAngle: 60 }, { text: "1,000", value: 1000, minAngle: 60, maxAngle: 120 }, { text: "500", value: 500, minAngle: 120, maxAngle: 180 }, { text: "2,000", value: 2000, minAngle: 180, maxAngle: 240 }, { text: "500", value: 500, minAngle: 240, maxAngle: 300 }, { text: "1,000", value: 1000, minAngle: 300, maxAngle: 360 } ]; let isSpinning = false; let currentRotation = 0; // 1. Function to check 24-hour limit (Once per day) function checkSpinLimit() { const lastSpin = localStorage.getItem('chubby_last_spin_timestamp'); const btn = document.getElementById('spinbtn'); const txt = document.getElementById('spinstatus'); const twentyFourHours = 86400000; // 24 * 60 * 60 * 1000 if (lastSpin) { const timePassed = new Date().getTime() - parseInt(lastSpin); if (timePassed < twentyFourHours) { btn.disabled = true; btn.style.opacity = "0.5"; // Calculate remaining time const timeLeft = twentyFourHours - timePassed; const hoursLeft = Math.floor(timeLeft / (1000 * 60 * 60)); const minutesLeft = Math.floor((timeLeft % (1000 * 60 * 60)) / (1000 * 60)); // Display countdown message in English txt.innerText = Already spun today! Try again in ${hoursLeft}h ${minutesLeft}m.; return false; } } btn.disabled = false; btn.style.opacity = "1"; return true; } // 2. Core function to rotate the wheel function startSpin() { if (!checkSpinLimit() || isSpinning) return; isSpinning = true; const wheel = document.getElementById("wheel"); const statusText = document.getElementById("spinstatus"); const prizeIndex = Math.floor(Math.random() * spinPrizes.length); const selectedPrize = spinPrizes[prizeIndex]; const randomSliceAngle = Math.floor(Math.random() * (selectedPrize.maxAngle - selectedPrize.minAngle)) + selectedPrize.minAngle; const extraSpins = 1800; currentRotation += extraSpins + (360 - randomSliceAngle); wheel.style.transition = "transform 4s cubic-bezier(0.25, 0.1, 0.25, 1)"; wheel.style.transform = rotate(${currentRotation}deg); setTimeout(() => { isSpinning = false; // Display winning result in English statusText.innerText = You won: ${selectedPrize.text}!; // Save current timestamp to local storage localStorage.setItem('chubby_last_spin_timestamp', new Date().getTime()); // Send the data to your Node.js backend sendPrizeToBackend(selectedPrize.value); // Lock the spin button instantly checkSpinLimit(); }, 4000); } // 3. Send the reward data to server.js function sendPrizeToBackend(prizeValue) { fetch('/api/save-prize', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prize: prizeValue, timestamp: new Date().getTime() }) }) .then(response => response.json()) .then(data => { console.log('Prize successfully saved on server:', data); }) .catch(error => { console.error('Error sending prize to server:', error); }); } // 4. Automatically run checks when the page loads window.onload = function() { checkSpinLimit(); };
